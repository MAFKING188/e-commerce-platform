% ================= OBJECTIVES =================
\begin{frame}{Strategic Project Objectives}

\begin{block}{Architectural Vision}
To engineer a high-integrity e-commerce engine capable of managing full transactional lifecycles in a scalable and secure environment.
\end{block}

\begin{itemize}
    \item \textbf{Transactional Flow:} Browse → Cart → Checkout → Order
    \item \textbf{Data Modeling:} Normalized relational structure (MLD)
    \item \textbf{Actor Separation:} Guest, Client, Admin, Server
    \item \textbf{Business Integrity:} Stock validation and atomic transactions
    \item \textbf{Security Layer:} Role-Based Access Control (RBAC)
\end{itemize}

\end{frame}

% ================= BUSINESS LOGIC =================
\begin{frame}{Business Logic Rules}

\begin{itemize}
\item Prevent checkout if cart is empty
\item Validate stock before order creation
\item Deduct stock after purchase
\item Prevent duplicate cart items (increment quantity)
\item Ensure atomic transactions
\end{itemize}

\end{frame}
